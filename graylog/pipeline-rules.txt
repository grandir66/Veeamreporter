// ============================================================================
// GRAYLOG PIPELINE RULES - Backup Monitor
// ============================================================================
//
// SETUP:
// 1. System > Pipelines > Manage rules > Create Rule (per ogni regola sotto)
// 2. System > Pipelines > Add new pipeline: "Backup Monitor Processing"
// 3. Collega le regole alla pipeline in ordine (Stage 0, Stage 1, etc.)
// 4. Collega la pipeline allo stream "All messages" o crea uno stream dedicato
//
// REGOLE: 8 totali (1 Stage 0 + 7 Stage 1)
//
// NOTA: Il campo "message" di Graylog contiene il formato:
//   MSGID - {"message_type":"...", ...}
// Le regole estraggono il JSON dopo il " - " con regex.
//
// ============================================================================

// ----------------------------------------------------------------------------
// RULE 1: Parse JSON dal messaggio syslog (Stage 0)
// ----------------------------------------------------------------------------
rule "backup_monitor_parse_json"
when
    has_field("message") AND
    contains(to_string($message.message), "\"message_type\":")
then
    // Estrai JSON dal messaggio (dopo il " - " del syslog RFC5424)
    let msg = to_string($message.message);
    let json_match = regex("(\\{.+\\})\\s*$", msg);
    let json_str = to_string(json_match["0"]);

    // Verifica che il JSON sia stato estratto
    let data = parse_json(json_str);

    // Campi base
    set_field("backup_message_type", data.message_type);
    set_field("backup_version", data.version);
    set_field("backup_timestamp", data.timestamp);
    set_field("backup_status", data.status);
    set_field("backup_agent_hostname", data.agent_hostname);

    // Client info
    set_field("client_code", data.client.code);
    set_field("client_name", data.client.name);
    set_field("client_site", data.client.site);

    // Salva JSON per le regole successive
    set_field("backup_json", json_str);
end

// ----------------------------------------------------------------------------
// RULE 2: Parse campi specifici VEEAM_JOB_RESULT (Stage 1)
// ----------------------------------------------------------------------------
rule "backup_monitor_veeam_job"
when
    has_field("backup_message_type") AND
    to_string($message.backup_message_type) == "VEEAM_JOB_RESULT"
then
    let data = parse_json(to_string($message.backup_json));

    set_field("job_id", data.job_id);
    set_field("job_name", data.job_name);
    set_field("job_type", data.job_type);
    set_field("job_start_time", data.start_time);
    set_field("job_end_time", data.end_time);
    set_field("job_duration_seconds", data.duration_seconds);
    set_field("job_duration_minutes", data.duration_minutes);
    set_field("job_result_message", data.result_message);
    set_field("data_size_bytes", data.data_size_bytes);
    set_field("data_size_gb", data.data_size_gb);
    set_field("transferred_bytes", data.transferred_bytes);
    set_field("transferred_gb", data.transferred_gb);
    set_field("objects_total", data.objects_total);
    set_field("objects_success", data.objects_success);
    set_field("objects_warning", data.objects_warning);
    set_field("objects_failed", data.objects_failed);
end

// ----------------------------------------------------------------------------
// RULE 4: Parse campi specifici REPOSITORY/DATASTORE/STORAGE STATUS (Stage 1)
// ----------------------------------------------------------------------------
rule "backup_monitor_storage_status"
when
    has_field("backup_message_type") AND
    (to_string($message.backup_message_type) == "VEEAM_REPOSITORY_STATUS" OR
     to_string($message.backup_message_type) == "PBS_DATASTORE_STATUS" OR
     to_string($message.backup_message_type) == "PVE_STORAGE_STATUS")
then
    let data = parse_json(to_string($message.backup_json));

    // Normalizza nome campo (repository_name, datastore_name o storage_name)
    let storage_name = data.repository_name;
    set_field("storage_name",
        regex_replace("^$", to_string(storage_name), to_string(data.datastore_name)));
    set_field("storage_name",
        regex_replace("^$", to_string($message.storage_name), to_string(data.storage_name)));

    set_field("storage_type", data.repository_type);
    set_field("storage_type",
        regex_replace("^$", to_string($message.storage_type), to_string(data.storage_type)));
    set_field("storage_path", data.repository_path);
    set_field("storage_total_bytes", data.total_bytes);
    set_field("storage_free_bytes", data.free_bytes);
    set_field("storage_used_percent", data.used_percent);
    set_field("storage_total_gb", data.total_gb);
    set_field("storage_free_gb", data.free_gb);
end

    set_field("server_memory_free_gb", data.memory_free_gb);
    set_field("server_memory_total_gb", data.memory_total_gb);

    // Licenza Veeam (solo per VEEAM_SERVER_STATUS)

    set_field("license_edition", data.license_edition);
